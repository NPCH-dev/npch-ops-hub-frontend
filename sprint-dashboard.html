SQL für Supabase einrichten (einmalig ausführen)
-- Supabase → SQL Editor → New Query → ausführen CREATE TABLE IF NOT EXISTS sprints ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name VARCHAR(100) NOT NULL, goal TEXT, start_date DATE NOT NULL, end_date DATE NOT NULL, status VARCHAR(20) DEFAULT 'planned', created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE IF NOT EXISTS tickets ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), ticket_id VARCHAR(20) UNIQUE NOT NULL, title VARCHAR(500) NOT NULL, description TEXT, status VARCHAR(20) DEFAULT 'ready', priority VARCHAR(10) DEFAULT 'medium', agent VARCHAR(50), sprint_id UUID REFERENCES sprints(id), story_points INTEGER DEFAULT 1, blocked_reason TEXT, created_at TIMESTAMPTZ DEFAULT NOW(), completed_at TIMESTAMPTZ, night_run_date DATE ); CREATE TABLE IF NOT EXISTS releases ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), version VARCHAR(50) NOT NULL, name VARCHAR(200), planned_date DATE, deployed_date DATE, status VARCHAR(20) DEFAULT 'planned', sprint_id UUID REFERENCES sprints(id), notes TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); CREATE TABLE IF NOT EXISTS daily_items ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), item_date DATE NOT NULL DEFAULT CURRENT_DATE, type VARCHAR(20), content TEXT NOT NULL, ticket_id VARCHAR(20), resolved BOOLEAN DEFAULT FALSE, created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE sprints ENABLE ROW LEVEL SECURITY; ALTER TABLE tickets ENABLE ROW LEVEL SECURITY; ALTER TABLE releases ENABLE ROW LEVEL SECURITY; ALTER TABLE daily_items ENABLE ROW LEVEL SECURITY; CREATE POLICY "public_all" ON sprints FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all" ON tickets FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all" ON releases FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all" ON daily_items FOR ALL USING (true) WITH CHECK (true);
